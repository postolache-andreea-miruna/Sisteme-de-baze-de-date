-- si in cele din urma sa se afiseze si competitia/competitiile cu cei mai multi participanti

SET SERVEROUTPUT ON

CREATE OR REPLACE PROCEDURE competitii_cheltuieli
IS
TYPE tip_record IS RECORD
(nume_comp COMPETITIE.nume%TYPE,
 nume_sp SPORTIV.nume%TYPE,
 prenume_sp SPORTIV.prenume%TYPE,
 taxa COMPETITIE.taxa_participare%TYPE,
 cod_sp SPORTIV.cod_sportiv%TYPE);
 
TYPE tablou_indexat IS TABLE OF tip_record INDEX BY PLS_INTEGER;
TYPE tablou_imbricat IS TABLE OF tip_record;
TYPE vector IS VARRAY(50) OF VARCHAR2(30);

t_indexat_aq tablou_indexat;
t_imbricat_bsc tablou_imbricat:= tablou_imbricat();
v_vector_competitii vector := vector();

suma_bsc NUMBER:=0;
suma_aq NUMBER:=0;
index_vector NUMBER:=1;

TYPE tablou_frecventa IS TABLE OF NUMBER INDEX BY VARCHAR2(30);
t_frecventa_competitii tablou_frecventa := tablou_frecventa();

v_cheie_frecventa VARCHAR2(30);
v_max_frecventa NUMBER := 0;
BEGIN

SELECT DISTINCT(COMPETITIE.nume), SPORTIV.nume, SPORTIV.prenume, COMPETITIE.taxa_participare, SPORTIV.cod_sportiv
BULK COLLECT INTO t_indexat_aq
FROM SPORTIV
JOIN ANTRENOR ON sportiv.cod_antrenor = antrenor.cod_antrenor
JOIN CLUB ON antrenor.cod_club = club.cod_club
JOIN PARTICIPA ON sportiv.cod_sportiv = participa.cod_sportiv
JOIN COMPETITIE ON participa.cod_competitie = competitie.cod_competitie
WHERE UPPER(club.nume_club) = 'AQUA TEAM'
AND UPPER(sportiv.tip_sportiv) <> 'SPORTIV LOT'
AND MONTHS_BETWEEN(SYSDATE,competitie.data_final)<=96 -- 8 ani
ORDER BY sportiv.cod_sportiv;


SELECT DISTINCT(COMPETITIE.nume), SPORTIV.nume, SPORTIV.prenume, COMPETITIE.taxa_participare, SPORTIV.cod_sportiv
BULK COLLECT INTO t_imbricat_bsc
FROM SPORTIV
JOIN ANTRENOR ON sportiv.cod_antrenor = antrenor.cod_antrenor
JOIN CLUB ON antrenor.cod_club = club.cod_club
JOIN PARTICIPA ON sportiv.cod_sportiv = participa.cod_sportiv
JOIN COMPETITIE ON participa.cod_competitie = competitie.cod_competitie
WHERE UPPER(club.nume_club) = 'BUCHAREST SPORT CLUB ELITE'
AND UPPER(sportiv.tip_sportiv) <> 'SPORTIV LOT'
AND MONTHS_BETWEEN(SYSDATE,competitie.data_final)<=96 -- 8 ani
ORDER BY sportiv.cod_sportiv;

DBMS_OUTPUT.PUT_LINE('CLUBUL AQUA TEAM');
DBMS_OUTPUT.NEW_LINE();


IF t_indexat_aq.COUNT >0 THEN
    FOR i IN t_indexat_aq.FIRST..t_indexat_aq.LAST LOOP
        IF t_indexat_aq.EXISTS(i) THEN
            suma_aq:=suma_aq+t_indexat_aq(i).taxa;
            v_vector_competitii.EXTEND;
            v_vector_competitii(index_vector) := t_indexat_aq(i).nume_comp;
            index_vector:=index_vector+1;
            IF i = t_indexat_aq.FIRST THEN
                DBMS_OUTPUT.PUT_LINE(t_indexat_aq(i).nume_sp||' '||t_indexat_aq(i).prenume_sp||' '||'a concurat la competitia: ');
                DBMS_OUTPUT.PUT_LINE(t_indexat_aq(i).nume_comp);
            END IF;
            IF i>=t_indexat_aq.FIRST+1 AND t_indexat_aq(i).cod_sp <> t_indexat_aq(i-1).cod_sp THEN 
                DBMS_OUTPUT.PUT_LINE(t_indexat_aq(i).nume_sp||' '||t_indexat_aq(i).prenume_sp||' '||'a concurat la competitia: ');
                DBMS_OUTPUT.PUT_LINE(t_indexat_aq(i).nume_comp);
            ELSIF i>=t_indexat_aq.FIRST+1 AND t_indexat_aq(i).cod_sp = t_indexat_aq(i-1).cod_sp THEN
                DBMS_OUTPUT.PUT_LINE(t_indexat_aq(i).nume_comp);
            END IF;
        END IF;
    END LOOP;
END IF;

DBMS_OUTPUT.NEW_LINE();
DBMS_OUTPUT.PUT_LINE('Clubul AQUA TEAM a cheltuit: '||suma_aq||' lei');
DBMS_OUTPUT.NEW_LINE();


DBMS_OUTPUT.PUT_LINE('CLUBUL BUCHAREST SPORT CLUB ELITE');
DBMS_OUTPUT.NEW_LINE();
IF t_imbricat_bsc.COUNT>0 THEN
    FOR i IN t_imbricat_bsc.FIRST..t_imbricat_bsc.LAST LOOP
        IF t_imbricat_bsc.EXISTS(i) THEN
            suma_bsc:=suma_bsc+t_imbricat_bsc(i).taxa;
            v_vector_competitii.EXTEND;
            v_vector_competitii(index_vector) := t_imbricat_bsc(i).nume_comp;
            index_vector:=index_vector+1;
            IF i = t_imbricat_bsc.FIRST THEN --primul rand din colectie
                DBMS_OUTPUT.PUT_LINE(t_imbricat_bsc(i).nume_sp||' '||t_imbricat_bsc(i).prenume_sp||' '||'a concurat la competitia: ');
                DBMS_OUTPUT.PUT_LINE(t_imbricat_bsc(i).nume_comp);
            END IF;
            IF i>=t_imbricat_bsc.FIRST+1 AND t_imbricat_bsc(i).cod_sp <> t_imbricat_bsc(i-1).cod_sp THEN--restul randurilor (daca am trecut la un alt sportiv)
                DBMS_OUTPUT.PUT_LINE(t_imbricat_bsc(i).nume_sp||' '||t_imbricat_bsc(i).prenume_sp||' '||'a concurat la competitia: ');
                DBMS_OUTPUT.PUT_LINE(t_imbricat_bsc(i).nume_comp);
            ELSIF i>=t_imbricat_bsc.FIRST+1 AND t_imbricat_bsc(i).cod_sp = t_imbricat_bsc(i-1).cod_sp THEN
                DBMS_OUTPUT.PUT_LINE(t_imbricat_bsc(i).nume_comp);

            END IF;
        END IF;
    END LOOP;
END IF;
DBMS_OUTPUT.NEW_LINE();
DBMS_OUTPUT.PUT_LINE('Clubul BUCHAREST SPORT CLUB ELITE a cheltuit: '||suma_bsc||' lei');

FOR i IN v_vector_competitii.FIRST..v_vector_competitii.LAST LOOP
    IF v_vector_competitii.EXISTS(i) THEN
      -- Check if the element already exists in the frequency array
      IF t_frecventa_competitii.EXISTS(v_vector_competitii(i)) THEN
        -- Increment the count if the element exists
        t_frecventa_competitii(v_vector_competitii(i)) := t_frecventa_competitii(v_vector_competitii(i)) + 1;
      ELSE
        -- Initialize count to 1 if the element doesn't exist
        t_frecventa_competitii(v_vector_competitii(i)) := 1;       
      END IF;
    END IF;
END LOOP;
DBMS_OUTPUT.NEW_LINE();
DBMS_OUTPUT.PUT_LINE('Participari la competitii:');
v_cheie_frecventa := t_frecventa_competitii.FIRST;
WHILE v_cheie_frecventa IS NOT NULL LOOP
      DBMS_OUTPUT.PUT_LINE('La competitia: ' || v_cheie_frecventa|| ' au participat '|| t_frecventa_competitii(v_cheie_frecventa) || ' sportivi din cele doua cluburi' );
      v_cheie_frecventa := t_frecventa_competitii.NEXT(v_cheie_frecventa);
END LOOP;

DBMS_OUTPUT.NEW_LINE();
DBMS_OUTPUT.PUT_LINE('Competitiile/competitia cu cei mai multi participanti de la cele doua cluburi');
v_cheie_frecventa := t_frecventa_competitii.FIRST;
WHILE v_cheie_frecventa IS NOT NULL LOOP
    IF t_frecventa_competitii.EXISTS(v_cheie_frecventa) THEN
        IF v_max_frecventa = 0 OR t_frecventa_competitii(v_cheie_frecventa) > v_max_frecventa THEN
            v_max_frecventa := t_frecventa_competitii(v_cheie_frecventa);
        END IF;
    END IF;
    v_cheie_frecventa := t_frecventa_competitii.NEXT(v_cheie_frecventa);
END LOOP;

v_cheie_frecventa := t_frecventa_competitii.FIRST;
WHILE v_cheie_frecventa IS NOT NULL LOOP
    IF t_frecventa_competitii(v_cheie_frecventa) = v_max_frecventa THEN
        DBMS_OUTPUT.PUT_LINE(v_cheie_frecventa);
    END IF;
    v_cheie_frecventa := t_frecventa_competitii.NEXT(v_cheie_frecventa);
   END LOOP;


END;
/
BEGIN
competitii_cheltuieli;
END;

/*
 CERINTA:
 Pentru fiecare club care are cel putin un antrenor care la randul sau are cel putin un sportiv care este sponsorizat 
in anul 2022 si care a fost medaliat la cel putin o competitie 
sa se afiseze numele clubului si sa se obtina lista antrenorilor (nume,prenume,salariu) care lucreaza in clubul respectiv ordonati alfabetic dupa nume.
Mai mult, sa se realizeze un numar de ordine pentru fiecare antrenor care se va reseta pentru fiecare club care respecta 
cerintele.
Pentru fiecare club care indeplineste conditia sa se afiseze numarul de antrenori,
valoarea lunara a veniturilor antrenorilor, valoarea medie a veniturilor antrenorilor
A se tine cont de faptul ca un club trebuie sa aiba cel putin un antrenor.
*/
--s-a folosit cursor explicit si cursor cu parametru
CREATE OR REPLACE PROCEDURE club_antr_sp
IS
i NUMBER;
venit_lunar NUMBER;

CURSOR cluburi IS SELECT club.nume_club, club.cod_club
               FROM CLUB
               JOIN ANTRENOR ON club.cod_club = antrenor.cod_club
               WHERE antrenor.cod_antrenor IN (SELECT antrenor.cod_antrenor
                                               FROM ANTRENOR
                                               JOIN SPORTIV ON antrenor.cod_antrenor = sportiv.cod_antrenor
                                               JOIN SPONSORIZEAZA ON sportiv.cod_sportiv = sponsorizeaza.cod_sportiv
                                               JOIN PARTICIPA ON sportiv.cod_sportiv = participa.cod_sportiv
                                               WHERE EXTRACT(YEAR FROM sponsorizeaza.inceput_colaborare) <= 2022 AND
                                                     EXTRACT(YEAR FROM sponsorizeaza.sfarsit_colaborare) >= 2022 AND
                                                     medaliat = 1
                                                GROUP BY antrenor.cod_antrenor
                                                );
CURSOR antr(v_club club.cod_club%TYPE) IS
    SELECT nume, prenume, salariu
    FROM ANTRENOR
    WHERE cod_club = v_club
    ORDER BY antrenor.nume;    
    
v_nume_club club.nume_club%TYPE;
v_cod_club club.cod_club%TYPE;
v_nume_antr antrenor.nume%TYPE;
v_prenume_antr antrenor.prenume%TYPE;
v_salariu antrenor.salariu%TYPE;
                                                
BEGIN

OPEN cluburi;
LOOP
    FETCH cluburi INTO v_nume_club, v_cod_club;
    EXIT WHEN cluburi%NOTFOUND;
    DBMS_OUTPUT.PUT_LINE('Clubul '||v_nume_club);
    DBMS_OUTPUT.PUT_LINE('-----------------------------------------------------------------------------------');
    
    i:=0;
    venit_lunar:=0;
    
    OPEN antr(v_cod_club);
    LOOP 
        FETCH antr INTO v_nume_antr, v_prenume_antr, v_salariu;
        EXIT WHEN antr%NOTFOUND;
        i:=i+1;
        venit_lunar:=venit_lunar + v_salariu;
        DBMS_OUTPUT.PUT_LINE(i||'. '||v_nume_antr||' '||v_prenume_antr);
    END LOOP;
    CLOSE antr;
    DBMS_OUTPUT.NEW_LINE;
    DBMS_OUTPUT.PUT_LINE('Numarul antrenorilor este de: '||i);
    DBMS_OUTPUT.PUT_LINE('Valoarea lunara a veniturilor antrenorilor este de: '||venit_lunar);
    IF i<>0 THEN
        DBMS_OUTPUT.PUT_LINE('Valoarea medie a veniturilor antrenorilor este de: '||venit_lunar/i);
    END IF;
END LOOP;
CLOSE cluburi;
END;
/
BEGIN
club_antr_sp;
END;

CREATE OR REPLACE FUNCTION nr_sportivi_nu_lot(v_nume COMPETITIE.nume%TYPE)
RETURN NUMBER
IS
v_cod_comp COMPETITIE.cod_competitie%TYPE;
v_nr_total_sp NUMBER;
v_nr_sp_nu_lot NUMBER;
v_data_start_comp COMPETITIE.data_start%TYPE;
nu_participari EXCEPTION;
competitie_viitor EXCEPTION;
BEGIN

DBMS_OUTPUT.PUT_LINE(v_nume);

SELECT cod_competitie
INTO v_cod_comp
FROM COMPETITIE
WHERE COMPETITIE.nume = v_nume; --NO DATA FOUND CAND INTRODUC UN NUME CARE NU EXISTA

SELECT data_start
INTO v_data_start_comp
FROM COMPETITIE
WHERE competitie.cod_competitie = v_cod_comp;

IF v_data_start_comp >= SYSDATE THEN
    RAISE competitie_viitor;
ELSE
    SELECT COUNT(DISTINCT cod_sportiv)--INTOARCE 0 PENTRU COMPETITIILE CARE EXISTA DAR CARE NU AU PARTICIPANTI
    INTO v_nr_total_sp
    FROM COMPETITIE
    JOIN PARTICIPA ON competitie.cod_competitie = participa.cod_competitie
    WHERE competitie.cod_competitie = v_cod_comp;

    IF v_nr_total_sp <>0 THEN

        SELECT COUNT(DISTINCT participa.cod_sportiv)
        INTO v_nr_sp_nu_lot
        FROM COMPETITIE
        JOIN PARTICIPA ON competitie.cod_competitie = participa.cod_competitie
        JOIN SPORTIV ON participa.cod_sportiv = sportiv.cod_sportiv
        WHERE sportiv.tip_sportiv <> 'SPORTIV LOT' AND
              competitie.cod_competitie = v_cod_comp;

        RETURN (v_nr_sp_nu_lot*100)/v_nr_total_sp;
    ELSE
        RAISE nu_participari;
    END IF;
END IF;

EXCEPTION
WHEN NO_DATA_FOUND THEN 
INSERT INTO MESAJE_INFO 
VALUES(SEQ_MESAJE_INFO.NEXTVAL,'Nu exista nicio competitie cu numele dat.','E',SYS.LOGIN_USER,SYSDATE);

COMMIT;
RAISE_APPLICATION_ERROR(-20001,'Aceasta competitie din trecut nu are participanti');
WHEN competitie_viitor THEN
INSERT INTO MESAJE_INFO 
VALUES(SEQ_MESAJE_INFO.NEXTVAL,'Aceasta competitie este programata in viitor','E',SYS.LOGIN_USER,SYSDATE);
COMMIT;
RAISE_APPLICATION_ERROR(-20002,'Aceasta competitie este programata in viitor');
WHEN OTHERS THEN 
INSERT INTO MESAJE_INFO 
VALUES(SEQ_MESAJE_INFO.NEXTVAL,'Alta eroare','E',SYS.LOGIN_USER,SYSDATE);
COMMIT;
RAISE_APPLICATION_ERROR(-20003,'Alta eroare');

PROCEDURE competitii_cheltuieli
IS
TYPE tip_record IS RECORD
(nume_comp COMPETITIE.nume%TYPE,
 nume_sp SPORTIV.nume%TYPE,
 prenume_sp SPORTIV.prenume%TYPE,
 taxa COMPETITIE.taxa_participare%TYPE,
 cod_sp SPORTIV.cod_sportiv%TYPE);
 
TYPE tablou_indexat IS TABLE OF tip_record INDEX BY PLS_INTEGER;
TYPE tablou_imbricat IS TABLE OF tip_record;
TYPE vector IS VARRAY(50) OF VARCHAR2(30);

t_indexat_aq tablou_indexat;
t_imbricat_bsc tablou_imbricat:= tablou_imbricat();
v_vector_competitii vector := vector();

suma_bsc NUMBER:=0;
suma_aq NUMBER:=0;
index_vector NUMBER:=1;

TYPE tablou_frecventa IS TABLE OF NUMBER INDEX BY VARCHAR2(30);
t_frecventa_competitii tablou_frecventa := tablou_frecventa();

v_cheie_frecventa VARCHAR2(30);
v_max_frecventa NUMBER := 0;
BEGIN

SELECT DISTINCT(COMPETITIE.nume), SPORTIV.nume, SPORTIV.prenume, COMPETITIE.taxa_participare, SPORTIV.cod_sportiv
BULK COLLECT INTO t_indexat_aq
FROM SPORTIV
JOIN ANTRENOR ON sportiv.cod_antrenor = antrenor.cod_antrenor
JOIN CLUB ON antrenor.cod_club = club.cod_club
JOIN PARTICIPA ON sportiv.cod_sportiv = participa.cod_sportiv
JOIN COMPETITIE ON participa.cod_competitie = competitie.cod_competitie
WHERE UPPER(club.nume_club) = 'AQUA TEAM'
AND UPPER(sportiv.tip_sportiv) <> 'SPORTIV LOT'
AND MONTHS_BETWEEN(SYSDATE,competitie.data_final)<=96 -- 8 ani
ORDER BY sportiv.cod_sportiv;


SELECT DISTINCT(COMPETITIE.nume), SPORTIV.nume, SPORTIV.prenume, COMPETITIE.taxa_participare, SPORTIV.cod_sportiv
BULK COLLECT INTO t_imbricat_bsc
FROM SPORTIV
JOIN ANTRENOR ON sportiv.cod_antrenor = antrenor.cod_antrenor
JOIN CLUB ON antrenor.cod_club = club.cod_club
JOIN PARTICIPA ON sportiv.cod_sportiv = participa.cod_sportiv
JOIN COMPETITIE ON participa.cod_competitie = competitie.cod_competitie
WHERE UPPER(club.nume_club) = 'BUCHAREST SPORT CLUB ELITE'
AND UPPER(sportiv.tip_sportiv) <> 'SPORTIV LOT'
AND MONTHS_BETWEEN(SYSDATE,competitie.data_final)<=96 -- 8 ani
ORDER BY sportiv.cod_sportiv;

DBMS_OUTPUT.PUT_LINE('CLUBUL AQUA TEAM');
DBMS_OUTPUT.NEW_LINE();


IF t_indexat_aq.COUNT >0 THEN
    FOR i IN t_indexat_aq.FIRST..t_indexat_aq.LAST LOOP

            v_vector_competitii(index_vector) := t_indexat_aq(i).nume_comp;
            index_vector:=index_vector+1;
            IF i = t_indexat_aq.FIRST THEN
                DBMS_OUTPUT.PUT_LINE(t_indexat_aq(i).nume_sp||' '||t_indexat_aq(i).prenume_sp||' '||'a concurat la competitia: ');
                DBMS_OUTPUT.PUT_LINE(t_indexat_aq(i).nume_comp);
            END IF;
            IF i>=t_indexat_aq.FIRST+1 AND t_indexat_aq(i).cod_sp <> t_indexat_aq(i-1).cod_sp THEN 
                DBMS_OUTPUT.PUT_LINE(t_indexat_aq(i).nume_sp||' '||t_indexat_aq(i).prenume_sp||' '||'a concurat la competitia: ');

            ELSIF i>=t_indexat_aq.FIRST+1 AND t_indexat_aq(i).cod_sp = t_indexat_aq(i-1).cod_sp THEN
                DBMS_OUTPUT.PUT_LINE(t_indexat_aq(i).nume_comp);
            END IF;
        END IF;
    END LOOP;
END IF;

DBMS_OUTPUT.NEW_LINE();
DBMS_OUTPUT.PUT_LINE('Clubul AQUA TEAM a cheltuit: '||suma_aq||' lei');
DBMS_OUTPUT.NEW_LINE();


DBMS_OUTPUT.PUT_LINE('CLUBUL BUCHAREST SPORT CLUB ELITE');
DBMS_OUTPUT.NEW_LINE();
IF t_imbricat_bsc.COUNT>0 THEN
    FOR i IN t_imbricat_bsc.FIRST..t_imbricat_bsc.LAST LOOP
        IF t_imbricat_bsc.EXISTS(i) THEN
            suma_bsc:=suma_bsc+t_imbricat_bsc(i).taxa;
            v_vector_competitii.EXTEND;
            v_vector_competitii(index_vector) := t_imbricat_bsc(i).nume_comp;
            index_vector:=index_vector+1;
            IF i = t_imbricat_bsc.FIRST THEN --primul rand din colectie

                DBMS_OUTPUT.PUT_LINE(t_imbricat_bsc(i).nume_comp);
            END IF;
            IF i>=t_imbricat_bsc.FIRST+1 AND t_imbricat_bsc(i).cod_sp <> t_imbricat_bsc(i-1).cod_sp THEN--restul randurilor (daca am trecut la un alt sportiv)
                DBMS_OUTPUT.PUT_LINE(t_imbricat_bsc(i).nume_sp||' '||t_imbricat_bsc(i).prenume_sp||' '||'a concurat la competitia: ');
                DBMS_OUTPUT.PUT_LINE(t_imbricat_bsc(i).nume_comp);
            ELSIF i>=t_imbricat_bsc.FIRST+1 AND t_imbricat_bsc(i).cod_sp = t_imbricat_bsc(i-1).cod_sp THEN

FOR i IN v_vector_competitii.FIRST..v_vector_competitii.LAST LOOP
    IF v_vector_competitii.EXISTS(i) THEN
      -- Check if the element already exists in the frequency array
      IF t_frecventa_competitii.EXISTS(v_vector_competitii(i)) THEN
        -- Increment the count if the element exists
        t_frecventa_competitii(v_vector_competitii(i)) := t_frecventa_competitii(v_vector_competitii(i)) + 1;
      ELSE
        -- Initialize count to 1 if the element doesn't exist
        t_frecventa_competitii(v_vector_competitii(i)) := 1;       
      END IF;
    END IF;

DBMS_OUTPUT.NEW_LINE();

v_cheie_frecventa := t_frecventa_competitii.FIRST;
WHILE v_cheie_frecventa IS NOT NULL LOOP
    IF t_frecventa_competitii.EXISTS(v_cheie_frecventa) THEN

END LOOP;

v_cheie_frecventa := t_frecventa_competitii.FIRST;
WHILE v_cheie_frecventa IS NOT NULL LOOP
    IF t_frecventa_competitii(v_cheie_frecventa) = v_max_frecventa THEN
        DBMS_OUTPUT.PUT_LINE(v_cheie_frecventa);
    END IF;
    v_cheie_frecventa := t_frecventa_competitii.NEXT(v_cheie_frecventa);
   END LOOP;
END;

i NUMBER;
venit_lunar NUMBER;

CURSOR cluburi IS SELECT club.nume_club, club.cod_club
               FROM CLUB
               JOIN ANTRENOR ON club.cod_club = antrenor.cod_club
               WHERE antrenor.cod_antrenor IN (SELECT antrenor.cod_antrenor
                                               FROM ANTRENOR
                                               JOIN SPORTIV ON antrenor.cod_antrenor = sportiv.cod_antrenor
                                               JOIN SPONSORIZEAZA ON sportiv.cod_sportiv = sponsorizeaza.cod_sportiv
                                               JOIN PARTICIPA ON sportiv.cod_sportiv = participa.cod_sportiv
                                               WHERE EXTRACT(YEAR FROM sponsorizeaza.inceput_colaborare) <= 2022 AND
                                                     EXTRACT(YEAR FROM sponsorizeaza.sfarsit_colaborare) >= 2022 AND
                                                     medaliat = 1
                                                GROUP BY antrenor.cod_antrenor
                                                );
CURSOR antr(v_club club.cod_club%TYPE) IS
    SELECT nume, prenume, salariu
    FROM ANTRENOR
    WHERE cod_club = v_club
    ORDER BY antrenor.nume;    
    
v_nume_club club.nume_club%TYPE;
v_cod_club club.cod_club%TYPE;
v_nume_antr antrenor.nume%TYPE;
v_prenume_antr antrenor.prenume%TYPE;
v_salariu antrenor.salariu%TYPE;
                                                
BEGIN

OPEN cluburi;
LOOP
    FETCH cluburi INTO v_nume_club, v_cod_club;
    EXIT WHEN cluburi%NOTFOUND;
    DBMS_OUTPUT.PUT_LINE('Clubul '||v_nume_club);
    DBMS_OUTPUT.PUT_LINE('-----------------------------------------------------------------------------------');
    
    i:=0;
    venit_lunar:=0;
    
    OPEN antr(v_cod_club);
    LOOP 
        FETCH antr INTO v_nume_antr, v_prenume_antr, v_salariu;
        EXIT WHEN antr%NOTFOUND;
        i:=i+1;
        venit_lunar:=venit_lunar + v_salariu;
        DBMS_OUTPUT.PUT_LINE(i||'. '||v_nume_antr||' '||v_prenume_antr);
    END LOOP;
    CLOSE antr;
    DBMS_OUTPUT.NEW_LINE;
    DBMS_OUTPUT.PUT_LINE('Numarul antrenorilor este de: '||i);
    DBMS_OUTPUT.PUT_LINE('Valoarea lunara a veniturilor antrenorilor este de: '||venit_lunar);
    IF i<>0 THEN
        DBMS_OUTPUT.PUT_LINE('Valoarea medie a veniturilor antrenorilor este de: '||venit_lunar/i);
    END IF;
END LOOP;
CLOSE cluburi;
END;

FUNCTION nr_sportivi_nu_lot(v_nume COMPETITIE.nume%TYPE)
RETURN NUMBER
IS
v_cod_comp COMPETITIE.cod_competitie%TYPE;
v_nr_total_sp NUMBER;
v_nr_sp_nu_lot NUMBER;
v_data_start_comp COMPETITIE.data_start%TYPE;
nu_participari EXCEPTION;
competitie_viitor EXCEPTION;
BEGIN

DBMS_OUTPUT.PUT_LINE(v_nume);

SELECT cod_competitie
INTO v_cod_comp
FROM COMPETITIE
WHERE COMPETITIE.nume = v_nume; --NO DATA FOUND CAND INTRODUC UN NUME CARE NU EXISTA

SELECT data_start
INTO v_data_start_comp
FROM COMPETITIE
WHERE competitie.cod_competitie = v_cod_comp;

IF v_data_start_comp >= SYSDATE THEN
    RAISE competitie_viitor;
ELSE
    SELECT COUNT(DISTINCT cod_sportiv)--INTOARCE 0 PENTRU COMPETITIILE CARE EXISTA DAR CARE NU AU PARTICIPANTI
    INTO v_nr_total_sp
    FROM COMPETITIE
    JOIN PARTICIPA ON competitie.cod_competitie = participa.cod_competitie
    WHERE competitie.cod_competitie = v_cod_comp;

    IF v_nr_total_sp <>0 THEN

        SELECT COUNT(DISTINCT participa.cod_sportiv)
        INTO v_nr_sp_nu_lot
        FROM COMPETITIE
        JOIN PARTICIPA ON competitie.cod_competitie = participa.cod_competitie
        JOIN SPORTIV ON participa.cod_sportiv = sportiv.cod_sportiv
        WHERE sportiv.tip_sportiv <> 'SPORTIV LOT' AND
              competitie.cod_competitie = v_cod_comp;

        RETURN (v_nr_sp_nu_lot*100)/v_nr_total_sp;
    ELSE
        RAISE nu_participari;
    END IF;
END IF;

EXCEPTION
WHEN NO_DATA_FOUND THEN 
INSERT INTO MESAJE_INFO 
VALUES(SEQ_MESAJE_INFO.NEXTVAL,'Nu exista nicio competitie cu numele dat.','E',SYS.LOGIN_USER,SYSDATE);
COMMIT;
RAISE_APPLICATION_ERROR(-20000,'Nu exista nicio competitie cu numele dat.');
WHEN nu_participari THEN
INSERT INTO MESAJE_INFO 
VALUES(SEQ_MESAJE_INFO.NEXTVAL,'Aceasta competitie din trecut nu are participanti','E',SYS.LOGIN_USER,SYSDATE);
COMMIT;
RAISE_APPLICATION_ERROR(-20001,'Aceasta competitie din trecut nu are participanti');
WHEN competitie_viitor THEN
INSERT INTO MESAJE_INFO 
VALUES(SEQ_MESAJE_INFO.NEXTVAL,'Aceasta competitie este programata in viitor','E',SYS.LOGIN_USER,SYSDATE);
COMMIT;
RAISE_APPLICATION_ERROR(-20002,'Aceasta competitie este programata in viitor');
WHEN OTHERS THEN 
INSERT INTO MESAJE_INFO 
VALUES(SEQ_MESAJE_INFO.NEXTVAL,'Alta eroare','E',SYS.LOGIN_USER,SYSDATE);
COMMIT;
RAISE_APPLICATION_ERROR(-20003,'Alta eroare');
END;

END pachet_proiect;
