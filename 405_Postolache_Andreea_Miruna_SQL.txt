--1.  
-- Sa se afiseze codul, numele, prenumele, indemnizatia sportivilor, daca sunt medaliati, numele, tipul, data de start si data de final a competitiei la care au participat.
-- Acestea vor fi afisate pentru sportivii care nu primesc indemnizatie (pentru cei care nu primesc indemnizatie se va trece 0 în coloana indemnizatie)
-- 				                       sau care participa la competitii care au în componenta primele 4 litere "CUPA"
-- 				                        si care în ambele cazuri trebuie sa fi participat la competitii care au avut loc cel mult acum 7 ani fata de data curenta.
-- De asemenea, numele si prenumele vor fi concatenate, iar coloana se va numi corespunzator.
-- Coloanele cod_sportiv, indemnizatie si numele competitiei se vor numi corespunzator.
-- Mai mult, aceste date trebuie ordonate dupa numele sportivului.

SELECT sp.cod_sportiv AS "Codul sportivului",CONCAT(sp.nume,'  ')||sp.prenume AS "Nume sportiv",NVL(sp.indemnizatie,0) "Indemnizatie", part.medaliat, comp.nume as "Nume competitie",comp.data_start,comp.data_final, t.tip_competitie
FROM SPORTIV sp INNER JOIN PARTICIPA part ON(sp.cod_sportiv=part.cod_sportiv)
                INNER JOIN COMPETITIE comp ON(comp.cod_competitie= part.cod_competitie)
                INNER JOIN TIP t ON(t.cod_tip=comp.cod_tip)
WHERE ((sp.indemnizatie = 0 OR INSTR(UPPER(comp.nume), 'CUPA')=1) AND MONTHS_BETWEEN(SYSDATE,comp.data_final)<=84) --7 ani
ORDER BY "Nume sportiv";

--2.
-- Pentru cluburile în care indemnizatia maxima este mai mare sau egala cu 500 si indemnizatia minima mai mica sau egala cu 300,
-- sa se obtina codul, numele acestor cluburi si indemnizatia maxima pe club, indemnizatia minima pe club si media indemnizatiilor pe club.
-- Nu se tine cont de valorile null de la sportivii care nu au indemnizatie.

SELECT cod_club, nume_club, MAX(indemnizatie)AS "INDEMNIZATIA MAXIMA", MIN(indemnizatie)AS "INDEMNIZATIA MINIMA", AVG(indemnizatie)AS "MEDIE INDEMNIZATII"
FROM SPORTIV  JOIN ANTRENOR USING(cod_antrenor)
             JOIN CLUB USING (cod_club)
GROUP BY cod_club,nume_club
HAVING MAX(indemnizatie) >= 500 AND MIN(indemnizatie)<=300;

--4.
-- Pentru fiecare club care are antrenori (sa se stie ca în aceasta diagrama un club trebuie sa aiba minimum un antrenor)
-- sa se afiseze codul clubului, suma totala a salariilor primite de antrenorii de la acel club
-- si într-o coloana cu numele STATUS BUGET SALARII sa se afiseze un mesaj care indica statusul bugetului pentru salarii:
-- pentru sumele mai mari de 1400 se va afisa mesajul: SUMA ALOCATA DE CLUB PENTRU A PLATI SALARIUL TUTUROR ANTRENORILOR ESTE MARE
-- pentru sumele între 1200 si 1400 se va afisa mesajul:SUMA ALOCATA DE CLUB PENTRU A PLATI SALARIUL TUTUROR ANTRENORILOR ESTE MEDIE
-- iar pentru restul se va afisa mesajul: SUMA ALOCATA DE CLUB PENTRU A PLATI SALARIUL TUTUROR ANTRENORILOR ESTE MICA

WITH SALARII AS(SELECT cod_club, SUM(salariu) SUMA
            FROM  ANTRENOR 
            GROUP BY cod_club
            ORDER BY cod_club
            )
select cod_club,CONCAT(TO_CHAR(SUMA),'  ')||'LEI' AS "SUMA",
CASE  WHEN SUMA>1400 THEN 'SUMA ALOCATA DE CLUB PENTRU A PLATI SALARIUL TUTUROR ANTRENORILOR ESTE MARE '
      WHEN SUMA< 1400 AND SUMA>1200 THEN 'SUMA ALOCATA DE CLUB PENTRU A PLATI SALARIUL TUTUROR ANTRENORILOR ESTE MEDIE'
      ELSE 'SUMA ALOCATA DE CLUB PENTRU A PLATI SALARIUL TUTUROR ANTRENORILOR ESTE MICA'
END AS "STATUS BUGET SALARII"
FROM SALARII;

-- Sa se afiseze numele, prenumele si stadiul sportivului pentru toti sportivii care sunt nascuti în aceeasi data ca antrenorul,

--6.
-- Sa se afiseze nivelele ierarhice ale tuturor antrenorilor nascuti inainte de 01-01-1991, numele, prenumele si experienta acestora,
-- dar si numele si prenumele sefului lor ( in cazul in care nu exista un sef, atunci in locul numelui se va trece simbolul "-").
-- De asemenea, numele si prenumele vor fi concatenate, iar coloana se va numi corespunzator.
-- Datele vor fi ordonate crescator in functie de nivelul din ierarhie si alfabetic in functie de numele antrenorului

SELECT LEVEL, CONCAT(a.nume,'  ')||a.prenume AS "NUME ANTRENOR",a.ani_experienta, 
        CASE
        WHEN NULLIF(sef.nume, '') IS NULL AND NULLIF(sef.prenume, '') IS NULL THEN '-'
        ELSE CONCAT(sef.nume, ' ')|| sef.prenume
        END AS "NUME SEF"
FROM ANTRENOR a
LEFT JOIN ANTRENOR sef ON a.cod_sef= sef.cod_antrenor
WHERE a.anul_nasterii < TO_DATE('01-01-1991','DD-MM-YYYY')
START WITH a.cod_sef IS NULL
CONNECT BY PRIOR a.cod_antrenor = a.cod_sef
ORDER BY LEVEL, "NUME ANTRENOR";

-- se face un tabel rez in care se face join intre tabele pentru a ajunge de la a lua numele si codul sportivului dorit
-- pana la a verifica ce rezultate au fost obtinute pe perioada sponsorizarii
WITH rez AS(SELECT rezultat.cod_proba codProba,proba.nume numeProba,sponsorizeaza.cod_sponsor codSponsor,timp_obtinut 
                FROM SPORTIV
                JOIN SPONSORIZEAZA ON sportiv.cod_sportiv = sponsorizeaza.cod_sportiv
                JOIN ISTORIC_SPORTIV ON sportiv.cod_sportiv = istoric_sportiv.cod_sportiv
                JOIN REZULTAT ON istoric_sportiv.cod_rezultat = rezultat.cod_rezultat
                JOIN PROBA ON rezultat.cod_proba = proba.cod_proba
            WHERE UPPER(sportiv.nume) = 'MARINCA' AND UPPER(sportiv.prenume)='RARES' AND sportiv.cod_sportiv = 1 AND
                  istoric_sportiv.data_rezultate_sportiv <= sponsorizeaza.sfarsit_colaborare AND
                  istoric_sportiv.data_rezultate_sportiv >= sponsorizeaza.inceput_colaborare)
                  
SELECT codProba, numeProba, COUNT(codSponsor)"NUMAR SPONSORI", MIN(timp_obtinut) 
FROM rez
WHERE timp_obtinut IN(SELECT MIN(timp_obtinut)FROM rez GROUP BY codProba)
GROUP BY codProba, numeProba
ORDER BY codProba;

--METODA CU WITH

WITH rez AS(
SELECT antrenor.nume nume,antrenor.prenume prenume, competitie.cod_competitie codcomp, participa.medaliat medaliati
FROM ANTRENOR
JOIN SPORTIV ON antrenor.cod_antrenor = sportiv.cod_antrenor
JOIN PARTICIPA ON sportiv.cod_sportiv = participa.cod_sportiv
JOIN COMPETITIE ON participa.cod_competitie = competitie.cod_competitie
JOIN ORAS ON competitie.cod_oras = oras.cod_oras
JOIN TARA ON oras.cod_tara = tara.cod_tara
WHERE UPPER(nume_tara) != 'ROMANIA')

SELECT nume||' '||prenume "NUME ANTRENOR", count(distinct codcomp) "NUMAR COMPETITII PARTICIPAT",SUM(medaliati)"NR. MEDALII COMPETITII STRAINATATE"
FROM rez
GROUP BY nume,prenume
ORDER BY "NUMAR COMPETITII PARTICIPAT" DESC;
