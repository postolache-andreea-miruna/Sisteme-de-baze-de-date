CREATE TABLE CLUB
(cod_club NUMBER(5),
nume_club VARCHAR2(50) CONSTRAINT nume_club_nn NOT NULL,
UNIQUE (nume_club)
);
ALTER TABLE CLUB 
ADD CONSTRAINT cod_club_pk PRIMARY KEY (cod_club);


CREATE TABLE ANTRENOR
(
cod_antrenor NUMBER(5),
nume VARCHAR2(25) CONSTRAINT nume_antrenor_nn NOT NULL,
prenume VARCHAR2(25) CONSTRAINT prenume_antrenor_nn NOT NULL,
anul_nasterii DATE CONSTRAINT nastere_nn NOT NULL,
gen CHAR(1) CONSTRAINT gen_nn NOT NULL, 
ani_experienta NUMBER(3),--poate fi null
salariu NUMBER(8,2) CONSTRAINT salariu_min CHECK (salariu>300.00),
cod_club NUMBER(5),
CONSTRAINT genul CHECK (gen='M' OR gen='F')
);
ALTER TABLE ANTRENOR 
ADD CONSTRAINT cod_antrenor_pk PRIMARY KEY(cod_antrenor);
ALTER TABLE ANTRENOR 
ADD CONSTRAINT cod_club_fk FOREIGN KEY(cod_club) REFERENCES CLUB (cod_club);
ALTER TABLE ANTRENOR
MODIFY cod_club number(5) NOT NULL;


CREATE TABLE NATIONALITATE
(
cod_nationalitate NUMBER(5),
nume_nationalitate VARCHAR2(15) CONSTRAINT nationalitate_sportiv_nn NOT NULL,
UNIQUE (nume_nationalitate),
CONSTRAINT cod_nationalitate_pk PRIMARY KEY (cod_nationalitate)
);


CREATE TABLE SPORTIV
(
cod_sportiv NUMBER(5),
gen CHAR(1) CONSTRAINT genul_sp_nn NOT NULL,
nume VARCHAR2(25) CONSTRAINT nume_sportiv_nn NOT NULL,
prenume VARCHAR2(25) CONSTRAINT prenume_sportiv_nn NOT NULL,
anul_nasterii DATE CONSTRAINT nastere_sp_nn NOT NULL,
cod_nationalitate NUMBER(5)CONSTRAINT nationalitate_sp_nn NOT NULL,
CONSTRAINT genul_sp CHECK (gen='M' OR gen='F'),
tip_sportiv VARCHAR2(25) CONSTRAINT tip_sportiv_nn NOT NULL,
indemnizatie NUMBER(5,1),
cod_antrenor NUMBER(5),
CONSTRAINT cod_sportiv_pk PRIMARY KEY (cod_sportiv),
CONSTRAINT cod_antrenor_fk FOREIGN KEY(cod_antrenor) REFERENCES ANTRENOR (cod_antrenor),
CONSTRAINT cod_nationalitate_fk FOREIGN KEY(cod_nationalitate) REFERENCES NATIONALITATE (cod_nationalitate)
);
ALTER TABLE SPORTIV
MODIFY cod_antrenor number(5) NOT NULL;


CREATE TABLE SPONSOR
(cod_sponsor NUMBER(5),
nume VARCHAR2(50) CONSTRAINT nume_sponsor_nn NOT NULL,
suma_contributie NUMBER(5,1)  CONSTRAINT suma_sponsor_nn NOT NULL,
UNIQUE (nume),
CONSTRAINT cod_sponsor_pk PRIMARY KEY (cod_sponsor)
);

CREATE TABLE SPONSORIZEAZA
(cod_sponsor NUMBER(5),
cod_sportiv NUMBER(5),
inceput_colaborare DATE CONSTRAINT inc_colab_nn NOT NULL,
sfarsit_colaborare DATE CONSTRAINT sf_colab_nn NOT NULL,
CONSTRAINT datele CHECK (inceput_colaborare<sfarsit_colaborare),
CONSTRAINT pk_cod_sponsor_cod_sportiv_colab PRIMARY KEY (cod_sponsor,cod_sportiv,inceput_colaborare),
CONSTRAINT cod_sponsor_fk FOREIGN KEY (cod_sponsor) REFERENCES SPONSOR (cod_sponsor),
CONSTRAINT cod_sportiv_fk FOREIGN KEY (cod_sportiv) REFERENCES SPORTIV (cod_sportiv)
);

CREATE TABLE TIP
(cod_tip NUMBER(5),
tip_competitie VARCHAR2(20) CONSTRAINT comp_tip_nn NOT NULL,
UNIQUE(tip_competitie)
);
ALTER TABLE TIP
ADD CONSTRAINT cod_tip_pk PRIMARY KEY (cod_tip);


CREATE TABLE TARA
(cod_tara CHAR(2),
nume_tara VARCHAR2(25) CONSTRAINT nume_tara_nn NOT NULL,
UNIQUE(nume_tara)
);
ALTER TABLE TARA
ADD CONSTRAINT cod_tara_pk PRIMARY KEY (cod_tara);


CREATE TABLE ORAS
(cod_oras NUMBER(5),
nume_oras VARCHAR2(25) CONSTRAINT nume_oras_nn NOT NULL,
cod_tara CHAR(2),
CONSTRAINT cod_oras_pk PRIMARY KEY (cod_oras),
CONSTRAINT cod_tara_fk FOREIGN KEY (cod_tara) REFERENCES TARA (cod_tara),
UNIQUE (nume_oras)
);
ALTER TABLE ORAS
MODIFY cod_tara char(2) NOT NULL;


CREATE TABLE COMPETITIE
(cod_competitie NUMBER(5),
nume  VARCHAR2(30) CONSTRAINT nume_comp_nn NOT NULL,
data_start DATE CONSTRAINT data_start_nn NOT NULL,
data_final DATE CONSTRAINT data_final_nn NOT NULL, 
taxa_participare NUMBER(4,1) CONSTRAINT taxa_nn NOT NULL,
cod_oras NUMBER(5),
cod_tip NUMBER(5),
CONSTRAINT cod_competitie_pk PRIMARY KEY (cod_competitie),
CONSTRAINT cod_tip_fk FOREIGN KEY (cod_tip) REFERENCES TIP (cod_tip),
CONSTRAINT cod_oras_fk FOREIGN KEY (cod_oras) REFERENCES ORAS (cod_oras),
CONSTRAINT date_concurs CHECK (data_start<data_final),
UNIQUE (nume)
);
ALTER TABLE COMPETITIE
MODIFY cod_oras number(5) NOT NULL;
ALTER TABLE COMPETITIE
MODIFY cod_tip number(5) NOT NULL;


CREATE TABLE PROBA
(
cod_proba NUMBER(5),
nume VARCHAR2(30) CONSTRAINT nume_proba_nn NOT NULL,
CONSTRAINT cod_proba_pk PRIMARY KEY (cod_proba),
UNIQUE (nume)
);


CREATE TABLE ISTORIC_SPORTIV
(
data_rezultate_sportiv DATE CONSTRAINT data_sportiv_rez NOT NULL,
cod_istoric  NUMBER(5),
cod_rezultat  NUMBER(5),
cod_sportiv  NUMBER(5),
CONSTRAINT pk_istoric_rezultat_sportiv primary key(cod_istoric, cod_rezultat,cod_sportiv),
CONSTRAINT cod_sportivul_fk FOREIGN KEY (cod_sportiv) REFERENCES SPORTIV (cod_sportiv)
);


CREATE TABLE REZULTAT
(
cod_rezultat NUMBER(5),
cod_proba NUMBER(5),
loc_clasament NUMBER(4) CONSTRAINT loc_cls_nn NOT NULL,
timp_obtinut NUMBER(5) CONSTRAINT timp_nn NOT NULL,--IN SECUNDE
cod_sportiv NUMBER(5),
cod_istoric NUMBER(5),
CONSTRAINT pk_rezultat PRIMARY KEY(cod_rezultat),
CONSTRAINT fk_istoric_rezultat_sportiv FOREIGN KEY(cod_istoric, cod_rezultat,cod_sportiv) REFERENCES ISTORIC_SPORTIV (cod_istoric,cod_rezultat,cod_sportiv),
CONSTRAINT fk_proba FOREIGN KEY(cod_proba) REFERENCES PROBA (cod_proba)
);


CREATE TABLE PARTICIPA
(
cod_sportiv NUMBER(5),
cod_competitie NUMBER(5),
cod_proba NUMBER(5),
medaliat NUMBER(1) CONSTRAINT medalie_nn NOT NULL,
CONSTRAINT medaliat_da_sau_nu CHECK (medaliat IN (1,0)),
CONSTRAINT pk_sportiv_rezultat_sportiv PRIMARY KEY (cod_sportiv,cod_competitie,cod_proba),
CONSTRAINT cod_sportivi_fk FOREIGN KEY (cod_sportiv) REFERENCES SPORTIV (cod_sportiv),
CONSTRAINT cod_compet_fk FOREIGN KEY (cod_competitie) REFERENCES COMPETITIE (cod_competitie),
CONSTRAINT cod_probele_fk FOREIGN KEY (cod_proba) REFERENCES PROBA (cod_proba)
);
